🏠 Home.tsx - 메인 페이지

역할: 필터링, 검색, 카드 그리드, 선택 기능 관리

주요 기능:

  • 카테고리별 필터링
  • 검색 기능
  • 카드 선택/해제
  • 전체 선택/해제 버튼

코드:

import { useState, useMemo, useCallback } from "react";
import { Search, X, CheckSquare, Square, Sparkles } from "lucide-react";
import { photocards, CATEGORIES, Category } from "@/data/photocards";
import PhotocardItem from "@/components/PhotocardItem";
import { cn } from "@/lib/utils";

export default function Home() {
  // 상태 관리
  const [activeCategory, setActiveCategory] = useState<Category>("All");
  const [searchQuery, setSearchQuery] = useState("");
  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());

  // 필터링된 카드 목록 (카테고리 + 검색어)
  const filteredCards = useMemo(() => {
    return photocards.filter((card) => {
      const matchCategory =
        activeCategory === "All" || card.category === activeCategory;
      const matchSearch =
        searchQuery.trim() === "" ||
        card.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
        card.category.toLowerCase().includes(searchQuery.toLowerCase());
      return matchCategory && matchSearch;
    });
  }, [activeCategory, searchQuery]);

  // 카드 선택/해제 토글
  const handleToggle = useCallback((id: string) => {
    setSelectedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) {
        next.delete(id);
      } else {
        next.add(id);
      }
      return next;
    });
  }, []);

  // 전체 선택
  const handleSelectAll = useCallback(() => {
    setSelectedIds(new Set(filteredCards.map((c) => c.id)));
  }, [filteredCards]);

  // 전체 해제
  const handleDeselectAll = useCallback(() => {
    setSelectedIds(new Set());
  }, []);

  const selectedCount = selectedIds.size;

  return (
    <div className="min-h-screen">
      {/* 헤더 */}
      <header className="sticky top-0 z-30 backdrop-blur-md bg-white/60">
        {/* 타이틀 및 카드 수 표시 */}
      </header>

      <main className="container py-6 space-y-5">
        {/* 카테고리 필터 */}
        <div className="overflow-x-auto pb-1">
          <div className="flex gap-2 w-max">
            {CATEGORIES.map((cat) => (
              <button
                key={cat}
                onClick={() => setActiveCategory(cat)}
                className={cn(
                  "filter-pill px-4 py-1.5 rounded-full text-sm font-medium",
                  activeCategory === cat
                    ? "bg-primary text-primary-foreground"
                    : "bg-white/70 text-foreground/70"
                )}
              >
                {cat}
              </button>
            ))}
          </div>
        </div>

        {/* 검색 + 액션 바 */}
        <div className="flex flex-col sm:flex-row gap-3">
          {/* 선택 카운터 */}
          <div className="flex items-center gap-1.5 text-sm">
            <span>선택된 카드: {selectedCount}개</span>
          </div>

          {/* 검색 입력 */}
          <div className="relative flex-1">
            <input
              type="text"
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              placeholder="카드 이름 또는 카테고리 검색..."
              className="w-full pl-9 pr-9 py-2 rounded-full"
            />
          </div>

          {/* 액션 버튼 */}
          <div className="flex gap-2">
            <button onClick={handleSelectAll}>전체 선택</button>
            <button onClick={handleDeselectAll} disabled={selectedCount === 0}>
              전체 해제
            </button>
          </div>
        </div>

        {/* 카드 그리드 - 모바일 3개, 데스크톱 6개 */}
        {filteredCards.length > 0 ? (
          <div className="grid grid-cols-3 lg:grid-cols-6 gap-3 sm:gap-4">
            {filteredCards.map((card, idx) => (
              <PhotocardItem
                key={card.id}
                card={card}
                selected={selectedIds.has(card.id)}
                onToggle={handleToggle}
                animationDelay={Math.min(idx * 35, 400)}
              />
            ))}
          </div>
        ) : (
          <div className="flex flex-col items-center justify-center py-24">
            <p>검색 결과가 없습니다</p>
          </div>
        )}
      </main>
    </div>
  );
}

🎴 PhotocardItem.tsx - 개별 카드 컴포넌트

역할: 포토카드 카드 UI 렌더링

주요 기능:

  • 카드 이미지 표시
  • 선택 시 흑백 + 투명도 70% 처리
  • 체크 배지 표시
  • 호버 시 카테고리 뱃지 표시

코드:

import { Check } from "lucide-react";
import { Photocard } from "@/data/photocards";
import { cn } from "@/lib/utils";

interface PhotocardItemProps {
  card: Photocard;
  selected: boolean;
  onToggle: (id: string) => void;
  animationDelay?: number;
}

export default function PhotocardItem({
  card,
  selected,
  onToggle,
  animationDelay = 0,
}: PhotocardItemProps) {
  return (
    <div
      className="card-animate"
      style={{ animationDelay: `${animationDelay}ms` }}
    >
      <button
        onClick={() => onToggle(card.id)}
        className={cn(
          "photocard-item group relative w-full rounded-xl overflow-hidden bg-white",
          "focus-visible:ring-2 focus-visible:ring-primary",
          "select-none"
        )}
        aria-pressed={selected}
        aria-label={`${card.name} 포토카드 ${selected ? "선택됨" : "선택 안됨"}`}
      >
        {/* 이미지 영역 */}
        <div className="relative aspect-[5/7] overflow-hidden">
          <img
            src={card.imageUrl}
            alt={card.name}
            className={cn(
              "w-full h-full object-cover transition-all duration-200",
              selected ? "grayscale opacity-70" : "grayscale-0 opacity-100"
            )}
            loading="lazy"
          />

          {/* 선택 오버레이 */}
          {selected && (
            <div className="absolute inset-0 bg-slate-900/10" />
          )}

          {/* 체크 배지 */}
          {selected && (
            <div className="check-badge absolute top-2 right-2 w-6 h-6 rounded-full bg-primary flex items-center justify-center shadow-md">
              <Check className="w-3.5 h-3.5 text-white" strokeWidth={3} />
            </div>
          )}

          {/* 호버 시 카테고리 뱃지 */}
          <div
            className={cn(
              "absolute bottom-0 left-0 right-0 px-2 py-1.5",
              "bg-gradient-to-t from-slate-900/60 to-transparent",
              "opacity-0 group-hover:opacity-100 transition-opacity duration-200"
            )}
          >
            <span className="text-white text-[10px] font-medium uppercase">
              {card.category}
            </span>
          </div>
        </div>

        {/* 카드 이름 */}
        <div className="px-2.5 py-2">
          <p
            className={cn(
              "text-[13px] font-medium text-center",
              "text-foreground/80 group-hover:text-foreground",
              selected && "text-foreground/50"
            )}
          >
            {card.name}
          </p>
          {card.year && (
            <p className="text-[11px] text-center text-muted-foreground mt-0.5">
              {card.year}
            </p>
          )}
        </div>
      </button>
    </div>
  );
}
